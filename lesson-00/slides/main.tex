\section{Course Overview}

\begin{frame}[c,allowframebreaks]{About the Course}
\textbf{Healthcare Data Privacy and Security} introduces students and healthcare professionals to the principles, policies, and technologies that protect health data within digital systems.

\bigskip
Topics covered:
\begin{itemize}
    \item Data privacy laws and ethical considerations
    \item Cybersecurity fundamentals and risk management
    \item Best practices for safeguarding electronic health information
\end{itemize}

\bigskip
\textbf{Schedule:} 10 lessons \texttimes{} 3h30m each
\end{frame}

\section{Program}

\begin{frame}[c,allowframebreaks]{Module 1 --- Basics}
\textit{(1 lesson)}
\begin{itemize}
    \item Importance of data protection in healthcare settings \cite{health_privacy_2025}
    \item Intro to data
    \begin{itemize}
        \item Definitions: Privacy, security, confidentiality, integrity, availability, \ldots
        \item Data lifecycle
    \end{itemize}
    \item Ethics in patient data protection
    \begin{itemize}
        \item Principles: autonomy, beneficence, justice, non-maleficence
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[c,allowframebreaks]{Module 2 --- Regulations and Compliance}
\textit{(2 lessons)}
\begin{itemize}
    \item International regulations: GDPR, HIPAA, \ldots
    \begin{itemize}
        \item Informed consent
        \item Types of health information (PHI, PII, \ldots)
        \item Anonymization and pseudonymization
        \item Data sharing and secondary use
    \end{itemize}
    \item Standards: ISO 27001, \ldots
    \item National laws
\end{itemize}
\end{frame}

\begin{frame}[c,allowframebreaks]{Module 3 --- Tech Background (1/2)}
\textit{(4 lessons)}
\begin{itemize}
    \item Health Information Systems
    \begin{itemize}
        \item EHR, LIS, RIS, PACS, Pharmacy systems
        \item Health Information Exchange (HIE), DHIS2
        \item APIs and interoperability (HL7/FHIR)
    \end{itemize}
    \item Intro to networking
    \item Encryption
    \begin{itemize}
        \item Encryption algorithms (basic overview)
        \item SSL/TLS, VPN, secure email systems
        \item Key management basics
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[c,allowframebreaks]{Module 3 --- Tech Background (2/2)}
\begin{itemize}
    \item Access control mechanisms
    \begin{itemize}
        \item Role-based access control (RBAC)
        \item Password policies and multi-factor authentication
    \end{itemize}
    \item Cyber threats
    \begin{itemize}
        \item Malware, ransomware, phishing, social engineering
        \item Insider threats
        \item Legacy systems and software vulnerabilities
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[c,allowframebreaks]{Module 4 --- System Security}
\textit{(2 lessons)}
\begin{itemize}
    \item Security Principles
    \begin{itemize}
        \item CIA triad (Confidentiality, Integrity, Availability)
        \item Defense in depth, zero trust, least privilege
        \item Security models and frameworks
    \end{itemize}
    \item Healthcare Security Applications
    \begin{itemize}
        \item EHR security; mobile health and telemedicine risks
        \item Cloud-based health systems
        \item Wearables, IoT health devices, DHIS2
        \item Audit logging and monitoring
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[c,allowframebreaks]{Module 5 --- Operational Security Management}
\textit{(2 lessons)}
\begin{itemize}
    \item Risk identification, analysis, and prioritization
    \item Threat modeling and security controls
    \item Mitigation strategies
    \item Security awareness and training
    \item Incident response lifecycle
    \item Backup, recovery, and business continuity
\end{itemize}
\end{frame}
